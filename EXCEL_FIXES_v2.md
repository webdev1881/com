## 🔧 Технические улучшения

### **Функция downloadTemplate():**
```javascript
// Динамически читает текущие данные
const currentStores = targetsData.value.storeTargets
const storeIds = Object.keys(currentStores) // Получаем все ID

// Создает заголовки на основе показателей
const indicators = Object.keys(targetsData.value.targetTree)
  .filter(key => key !== 'turnover')

// Генерирует строки для всех магазинов
storeIds.forEach(storeId => {
  const storeData = currentStores[storeId]
  const row = [
    storeId, // ID
    ...indicators.map(key => storeData[key] || 0), // Значения
    storeData.store || `Магазин ${storeId}` // Название
  ]
  targetsData_array.push(row)
})
```

### **Функция parseExcelData():**
```javascript
// Интеллектуальное сопоставление названий
const nameToKeyMapping = {
  'Списання': 'losses',
  'Нестачі': 'shortages', 
  'ФОП': 'fop',
  'Повернення': 'shiftRemainder',
  'Непровед.': 'unprocessed'
}

// Поиск в targetTree если не найдено в маппинге
for (const [treeKey, treeValue] of Object.entries(targetsData.value.targetTree)) {
  if (headerName.includes(treeValue.name) || treeValue.name.includes(headerName)) {
    key = treeKey
    break
  }
}
```
